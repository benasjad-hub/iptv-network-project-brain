// AFTER
  const safePages = useMemo(() => {
    if (!hasDomain || !site.domain) return [];
    const list = Array.isArray(pages) ? pages : [];

    let matchedHosts: string[] = [];
    if (
      trustedAliases &&
      typeof trustedAliases === 'object' &&
      !Array.isArray(trustedAliases) &&
      typeof trustedAliases.siteSlug === 'string' &&
      typeof trustedAliases.domain === 'string' &&
      Array.isArray(trustedAliases.hosts) &&
      trustedAliases.siteSlug === site.slug &&
      trustedAliases.domain === site.domain
    ) {
      matchedHosts = trustedAliases.hosts;
    }

    const clientValidatedAliases = validateHostAliases(site.domain, matchedHosts);
    return list.filter((p) =>
      isCanonicalSitePageEntry(p, site.domain, clientValidatedAliases)
    );

  test('1. actual registered www alias accepted', () => {
    const domain = 'vikingiptv.tv';
    const siteSlug = 'vikingiptvtv';
    const resolvedAliases = resolveTrustedHostAliases(domain, siteSlug, MOCK_HOST_TO_SITE);

    assert.ok(guide);
    assert.equal(guide.url, 'https://www.vikingiptv.tv/guides/installation');
    assert.equal(guide.path, '/guides/installation');
