### Programming Proposal: Server-Side Host Alias Resolution & Client Provenance Gate

This proposal resolves the integration gap where sitemaps for domains such as `vikingiptv.tv` emit canonical `www.vikingiptv.tv` URLs. Host aliases are strictly bound server-side from `HOST_TO_SITE` using exact site slug identity, apex/www pairing on the same base host, and explicit client provenance checks. Legacy 2-argument helper invocations retain their strict single-domain behavior.

function validateHostAliases(cleanDomain, aliases) {
  if (!aliases) return [];
  const list = Array.isArray(aliases) ? aliases : [aliases];
  const valid = [];
  for (const alias of list) {
    if (!alias || typeof alias !== 'string') continue;
    let cleanAlias;
    try {
      cleanAlias = assertSiteDomain(alias);
    } catch {
      continue;
    }
    if (cleanAlias === cleanDomain) continue;
    if (cleanAlias.endsWith('.workers.dev') || cleanAlias === 'workers.dev') {
      continue;
    }
    const isPair =
      cleanAlias === `www.${cleanDomain}` ||
      cleanDomain === `www.${cleanAlias}`;
    if (!isPair) {
      continue;
    }
    if (!valid.includes(cleanAlias)) {
      valid.push(cleanAlias);

  return valid;

function resolveTrustedHostAliases(domain, siteSlug, hostToSiteMap) {
  if (!domain || typeof domain !== 'string') return [];
  if (!siteSlug || typeof siteSlug !== 'string') return [];
  if (!hostToSiteMap || typeof hostToSiteMap !== 'object') return [];

  let cleanDomain;
  try {
    cleanDomain = assertSiteDomain(domain);
  } catch {
    return [];
  }

  if (cleanDomain.endsWith('.workers.dev') || cleanDomain === 'workers.dev') {
    return [];
  }

  if (
    !Object.prototype.hasOwnProperty.call(hostToSiteMap, cleanDomain) ||
    hostToSiteMap[cleanDomain] !== siteSlug
  ) {
    return [];
  }

  let candidate;
  if (cleanDomain.startsWith('www.')) {
    candidate = cleanDomain.slice(4);
  } else {
    candidate = `www.${cleanDomain}`;
  }

  try {
    candidate = assertSiteDomain(candidate);
  } catch {
    return [];
  }

  if (candidate.endsWith('.workers.dev') || candidate === 'workers.dev') {
    return [];
  }

  if (
    !Object.prototype.hasOwnProperty.call(hostToSiteMap, candidate) ||
    hostToSiteMap[candidate] !== siteSlug
  ) {
    return [];
  }

  return [candidate];
}

const { test, describe } = require('node:test');
const assert = require('node:assert/strict');
const {
  isCanonicalSitePageEntry,
  normalizeSitemapUrls,
  resolveTrustedHostAliases,
  validateHostAliases,
} = require('../scripts/studio/site-page-catalog.cjs');

const MOCK_HOST_TO_SITE = {
  'vikingiptv.tv': 'vikingiptvtv',
  'www.vikingiptv.tv': 'vikingiptvtv',
  'vikingiptvtv-cf.bendaoud-fst.workers.dev': 'vikingiptvtv',
  'vikingiptv.ca': 'vikingiptv',
  'www.vikingiptv.ca': 'vikingiptv',
  'polderiptv.nl': 'polderiptv',
  'www.polderiptv.nl': 'polderiptv',

describe('site-page-catalog host aliases & client provenance gate', () => {
  test('1. actual registered www alias accepted', () => {
    const domain = 'vikingiptv.tv';
    const siteSlug = 'vikingiptvtv';
    const resolvedAliases = resolveTrustedHostAliases(domain, siteSlug, MOCK_HOST_TO_SITE);
    assert.deepEqual(resolvedAliases, ['www.vikingiptv.tv']);

    const xml = `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
  <loc>https://www.vikingiptv.tv/</loc>
  <loc>https://www.vikingiptv.tv/tarifs</loc>
  <loc>https://www.vikingiptv.tv/guides/installation</loc>
</urlset>`;

    const pages = normalizeSitemapUrls(xml, domain, resolvedAliases);
    assert.equal(pages.length, 3);
    assert.equal(pages[0].url, 'https://www.vikingiptv.tv/');
    assert.equal(pages[0].path, '/');
    assert.equal(pages[0].kind, 'home');
    assert.equal(pages[1].url, 'https://www.vikingiptv.tv/tarifs');
    assert.equal(pages[1].path, '/tarifs');
    assert.equal(pages[2].url, 'https://www.vikingiptv.tv/guides/installation');
    assert.equal(pages[2].path, '/guides/installation');
    assert.equal(pages[2].kind, 'article');

    for (const page of pages) {
      assert.equal(isCanonicalSitePageEntry(page, domain, resolvedAliases), true);
    }
  });

  test('2. legacy strict preserved', () => {
    const xml = `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
  <loc>https://www.vikingiptv.tv/tarifs</loc>
  <loc>https://vikingiptv.tv/contact</loc>
</urlset>`;

    // Calling legacy 2-arg helper drops www URLs and accepts strictly exact-domain
    const pages = normalizeSitemapUrls(xml, 'vikingiptv.tv');
    assert.equal(pages.length, 1);
    assert.equal(pages[0].url, 'https://vikingiptv.tv/contact');

    const wwwEntry = {
      url: 'https://www.vikingiptv.tv/tarifs',
      path: '/tarifs',
      kind: 'page',
      label: 'Tarifs',
    };
    // 2-arg validation strictly rejects any alias
    assert.equal(isCanonicalSitePageEntry(wwwEntry, 'vikingiptv.tv'), false);

    const apexEntry = {
      url: 'https://vikingiptv.tv/contact',
      path: '/contact',
      kind: 'page',
      label: 'Contact',

    assert.equal(isCanonicalSitePageEntry(apexEntry, 'vikingiptv.tv'), true);
  });

  test('3. crosssite/substr/workers.dev/unknown aliases rejected', () => {

<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
