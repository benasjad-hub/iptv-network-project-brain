```javascript
  if (kind === 'home' || pathname === '/') return 'Accueil';
  const segments = pathname.split('/').filter(Boolean);
  const last = segments[segments.length - 1] || '';
  try {
    const decoded = decodeURIComponent(last).replace(/[-_]+/g, ' ').trim();
    if (decoded) return decoded.charAt(0).toUpperCase() + decoded.slice(1);
  } catch {

}
```

**AFTER:**
```javascript
function deriveLabel(pathname, kind) {
  if (kind === 'home' || pathname === '/') return 'Accueil';
  const segments = pathname.split('/').filter(Boolean);
  const last = segments[segments.length - 1] || '';
  let label = '';
  try {
    const decoded = decodeURIComponent(last).replace(/[-_]+/g, ' ').trim();
    if (decoded) label = decoded.charAt(0).toUpperCase() + decoded.slice(1);
  } catch {
    // fallback to clean unescaped
  }
  if (!label) {
    const clean = last.replace(/[-_]+/g, ' ').trim();
    label = clean ? clean.charAt(0).toUpperCase() + clean.slice(1) : pathname;
  }
  return label.slice(0, 200);
}
```
